# Boot spi DMA+CRC shell, trigger test A (SPI0 master + SPI1 slave)
# Host times mark-to-mark for cross-check against DSP-reported rate.
dsp:reset
dsp:uart_open
dsp:boot ldr=@main.ldr
delay ms=800
mark tag=start
dsp:uart_write data="a\r"
dsp:uart_expect sentinel="throughput:" timeout_ms=10000
mark tag=end
delay ms=500
dsp:uart_close
