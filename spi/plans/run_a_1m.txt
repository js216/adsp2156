# Test A with 1 MiB transfer (no jumpers -> expect early abort).
dsp:reset
dsp:uart_open
dsp:boot ldr=@main.ldr
delay ms=600
mark tag=cfg
dsp:uart_write data="n 1m\r"
delay ms=200
mark tag=start
dsp:uart_write data="a\r"
dsp:uart_expect sentinel="throughput:" timeout_ms=30000
mark tag=end
delay ms=300
dsp:uart_close
