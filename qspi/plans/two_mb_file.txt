# 2 MiB via blob file, 4B prefix, chunk=65526.
dsp:reset
dsp:uart_open
dsp:boot ldr=@main.ldr
delay ms=500
dsp:uart_write data="m4\r"
delay ms=50
dsp:uart_write data="?\r"
delay ms=200
mark tag=s
dsp:qspi_write data=@prbs2m.bin clk_div=8 mode=4 chunk_size=65526 prefix=@safety4.bin
mark tag=e
delay ms=200
dsp:uart_write data="i\r"
dsp:uart_expect sentinel="sum=0x" timeout_ms=10000
dsp:uart_close
