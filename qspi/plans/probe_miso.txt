# Drive each PA0..PA5 HIGH then LOW, FT4222 SingleRead after each,
# find which DSP pin is actually wired to master MISO.

dsp:reset
dsp:uart_open
dsp:boot ldr=@main.ldr
delay ms=500

dsp:uart_write data="g 0 0\r"
dsp:uart_expect sentinel="gpio PA" timeout_ms=1000
mark tag=pa0_lo
dsp:qspi_read n=4 clk_div=8 mode=1 chunk_size=4

dsp:uart_write data="g 1 0\r"
dsp:uart_expect sentinel="gpio PA" timeout_ms=1000
mark tag=pa1_lo
dsp:qspi_read n=4 clk_div=8 mode=1 chunk_size=4

dsp:uart_write data="g 2 0\r"
dsp:uart_expect sentinel="gpio PA" timeout_ms=1000
mark tag=pa2_lo
dsp:qspi_read n=4 clk_div=8 mode=1 chunk_size=4

dsp:uart_write data="g 3 0\r"
dsp:uart_expect sentinel="gpio PA" timeout_ms=1000
mark tag=pa3_lo
dsp:qspi_read n=4 clk_div=8 mode=1 chunk_size=4

dsp:uart_write data="g 4 0\r"
dsp:uart_expect sentinel="gpio PA" timeout_ms=1000
mark tag=pa4_lo
dsp:qspi_read n=4 clk_div=8 mode=1 chunk_size=4

dsp:uart_write data="g 5 0\r"
dsp:uart_expect sentinel="gpio PA" timeout_ms=1000
mark tag=pa5_lo
dsp:qspi_read n=4 clk_div=8 mode=1 chunk_size=4

dsp:uart_close
