# Cycle DSP PA0 state via GPIO force, poll bench MCU pin monitor
# to find which STM32 pin is wired to DSP MISO.
dsp:reset
dsp:uart_open
dsp:boot ldr=@main.ldr
delay ms=500

# Baseline: SPI running in slave RX, MISO pad is driven by EMISO
bench_mcu:send data="p\r"
delay ms=100

# Force PA0 LOW via GPIO (disables SPI to do this)
dsp:uart_write data="g 0 0\r"
dsp:uart_expect sentinel="gpio PA0" timeout_ms=1000
delay ms=100
bench_mcu:send data="p\r"
delay ms=100

# Force PA0 HIGH
dsp:uart_write data="g 0 1\r"
dsp:uart_expect sentinel="gpio PA0" timeout_ms=1000
delay ms=100
bench_mcu:send data="p\r"
delay ms=100

# Force PA1 LOW (MOSI)
dsp:uart_write data="g 1 0\r"
dsp:uart_expect sentinel="gpio PA1" timeout_ms=1000
delay ms=100
bench_mcu:send data="p\r"
delay ms=100

# Force PA4 LOW (CLK)
dsp:uart_write data="g 4 0\r"
dsp:uart_expect sentinel="gpio PA4" timeout_ms=1000
delay ms=100
bench_mcu:send data="p\r"
delay ms=100

# Force PA5 LOW (CS)
dsp:uart_write data="g 5 0\r"
dsp:uart_expect sentinel="gpio PA5" timeout_ms=1000
delay ms=100
bench_mcu:send data="p\r"

dsp:uart_close
