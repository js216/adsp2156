# Slave TX bring-up.  DSP stages a pattern in TFIFO (polled 'q'
# command), master reads via SingleReadWrite / MultiRead, results
# land in dsp.qspi_read stream.  Currently failing: slave shifter
# consumes TFIFO but MISO pin stays 0xFF on master side.  Used
# as the single regression test for slave TX work-in-progress.

dsp:reset
dsp:uart_open
dsp:boot ldr=@main.ldr
delay ms=500
dsp:uart_write data="?\r"
delay ms=200
dsp:uart_write data="a 0\r"
delay ms=50
# 'q' drops CTL.EN, sets TEN+REN+TFIFO, re-enables.  Shell prints
# "polled tx loaded ..." line we can wait on.
dsp:uart_write data="q\r"
dsp:uart_expect sentinel="polled tx loaded" timeout_ms=5000
delay ms=50
mark tag=xfer_start
# Master reads 16 bytes.  In single slave mode, FT4222 should see
# the 4 words from TFIFO (0xAA55AA55 DEADBEEF CAFEBABE 12345678).
dsp:qspi_read n=16 clk_div=8 mode=1 chunk_size=16
mark tag=xfer_end
delay ms=50
dsp:uart_write data="x\r"
dsp:uart_expect sentinel="tx_stat" timeout_ms=5000
dsp:uart_close
