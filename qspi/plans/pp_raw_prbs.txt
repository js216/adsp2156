# Baseline: FT4222 master sends pure PRBS(123) 40 bytes in quad
# mode, 8 times.  DSP drains each burst, prints cksum.  If PRBS
# arrives intact, every cksum matches host's running_xor of the
# same 40-byte stream.  Used as control for first-byte-hazard scan.

dsp:reset
dsp:uart_open
dsp:boot ldr=@main.ldr
delay ms=500
dsp:uart_write data="m4\r"
delay ms=50
dsp:uart_write data="a 0\r"
delay ms=50
dsp:uart_write data="?\r"
delay ms=200

# 8 back-to-back bursts of PRBS seed 123, 40 bytes each, quad mode.
# qspi_write_prbs generates the bytes plugin-side; no blob needed.
dsp:qspi_write_prbs seed=123 n=40 clk_div=8 mode=4
delay ms=10
dsp:uart_write data="i\r"
dsp:qspi_write_prbs seed=123 n=40 clk_div=8 mode=4
delay ms=10
dsp:uart_write data="i\r"
dsp:qspi_write_prbs seed=123 n=40 clk_div=8 mode=4
delay ms=10
dsp:uart_write data="i\r"
dsp:qspi_write_prbs seed=123 n=40 clk_div=8 mode=4
delay ms=10
dsp:uart_write data="i\r"
dsp:qspi_write_prbs seed=123 n=40 clk_div=8 mode=4
delay ms=10
dsp:uart_write data="i\r"
dsp:qspi_write_prbs seed=123 n=40 clk_div=8 mode=4
delay ms=10
dsp:uart_write data="i\r"
dsp:qspi_write_prbs seed=123 n=40 clk_div=8 mode=4
delay ms=10
dsp:uart_write data="i\r"
dsp:qspi_write_prbs seed=123 n=40 clk_div=8 mode=4
delay ms=10
dsp:uart_write data="i\r"
dsp:uart_expect sentinel="sum=0x" timeout_ms=5000
dsp:uart_close
