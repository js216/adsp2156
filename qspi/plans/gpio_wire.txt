# PA0 wire test.  DSP drives PA0 LOW then HIGH via raw GPIO (FER=0,
# DIR=1, DATA bit toggled via 'g0'/'g1').  Master reads via
# qspi_read; if the wire is connected, lo_xfer bytes = 0x00 and
# hi_xfer bytes = 0xFF.  If both read 0xFF, MISO trace is broken
# at the board level.

dsp:reset
dsp:uart_open
dsp:boot ldr=@main.ldr
delay ms=500
dsp:uart_write data="?\r"
delay ms=200
dsp:uart_write data="a 0\r"
delay ms=50
# PA0 = LOW.
dsp:uart_write data="g0\r"
dsp:uart_expect sentinel="gpio PA0" timeout_ms=1000
delay ms=20
mark tag=lo_xfer
dsp:qspi_read n=16 clk_div=8 mode=1
# PA0 = HIGH.
dsp:uart_write data="g1\r"
dsp:uart_expect sentinel="gpio PA0" timeout_ms=1000
delay ms=20
mark tag=hi_xfer
dsp:qspi_read n=16 clk_div=8 mode=1
dsp:uart_close
