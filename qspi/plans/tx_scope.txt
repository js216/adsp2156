# Stage all-zero TFIFO on slave, master SingleReads 64 bytes at
# slow clock so the CS-low window is visible on the scope.
# If slave shifter engages, C4 (MISO) drops LOW for the window.
dsp:reset
dsp:uart_open
dsp:boot ldr=@main.ldr
delay ms=500
dsp:uart_write data="m1\r"
delay ms=50
dsp:uart_write data="t 0 64\r"
dsp:uart_expect sentinel="tx queued" timeout_ms=2000
delay ms=20
mark tag=read
dsp:qspi_read n=64 clk_div=512 mode=1 chunk_size=64
mark tag=read_end
delay ms=100
scope:capture chans="C1,C2,C3,C4"
dsp:uart_close
