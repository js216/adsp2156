# Throughput sweep using plugin-side PRBS with 4-byte safety prefix.
# Seed substituted via sed (__SEED__).

dsp:reset
dsp:uart_open
dsp:boot ldr=@main.ldr
delay ms=500

# m1 ----------------------------------------------------------------
dsp:uart_write data="m1\r"
delay ms=50
dsp:uart_write data="?\r"
delay ms=200
mark tag=m1_div8_start
dsp:qspi_write_prbs seed=__SEED__ n=1048576 clk_div=8 mode=1 chunk_size=65535
mark tag=m1_div8_end
delay ms=50
dsp:uart_write data="i\r"
delay ms=2000

mark tag=m1_div4_start
dsp:qspi_write_prbs seed=__SEED__ n=1048576 clk_div=4 mode=1 chunk_size=65535
mark tag=m1_div4_end
delay ms=50
dsp:uart_write data="i\r"
delay ms=2000

mark tag=m1_div2_start
dsp:qspi_write_prbs seed=__SEED__ n=1048576 clk_div=2 mode=1 chunk_size=65535
mark tag=m1_div2_end
delay ms=50
dsp:uart_write data="i\r"
delay ms=2000

# m2 ----------------------------------------------------------------
dsp:uart_write data="m2\r"
delay ms=50
dsp:uart_write data="?\r"
delay ms=200
mark tag=m2_div8_start
dsp:qspi_write_prbs seed=__SEED__ n=1048576 clk_div=8 mode=2 chunk_size=65524 prefix=@safety4.bin
mark tag=m2_div8_end
delay ms=50
dsp:uart_write data="i\r"
delay ms=2000

mark tag=m2_div4_start
dsp:qspi_write_prbs seed=__SEED__ n=1048576 clk_div=4 mode=2 chunk_size=65524 prefix=@safety4.bin
mark tag=m2_div4_end
delay ms=50
dsp:uart_write data="i\r"
delay ms=2000

mark tag=m2_div2_start
dsp:qspi_write_prbs seed=__SEED__ n=1048576 clk_div=2 mode=2 chunk_size=65524 prefix=@safety4.bin
mark tag=m2_div2_end
delay ms=50
dsp:uart_write data="i\r"
delay ms=2000

# m4 ----------------------------------------------------------------
dsp:uart_write data="m4\r"
delay ms=50
dsp:uart_write data="?\r"
delay ms=200
mark tag=m4_div8_start
dsp:qspi_write_prbs seed=__SEED__ n=1048576 clk_div=8 mode=4 chunk_size=65524 prefix=@safety4.bin
mark tag=m4_div8_end
delay ms=50
dsp:uart_write data="i\r"
delay ms=2000

mark tag=m4_div4_start
dsp:qspi_write_prbs seed=__SEED__ n=1048576 clk_div=4 mode=4 chunk_size=65524 prefix=@safety4.bin
mark tag=m4_div4_end
delay ms=50
dsp:uart_write data="i\r"
delay ms=2000

mark tag=m4_div2_start
dsp:qspi_write_prbs seed=__SEED__ n=1048576 clk_div=2 mode=4 chunk_size=65524 prefix=@safety4.bin
mark tag=m4_div2_end
delay ms=50
dsp:uart_write data="i\r"
delay ms=2000

dsp:uart_write data="a 1\r"
dsp:uart_expect sentinel="auto=on" timeout_ms=5000
dsp:uart_close
