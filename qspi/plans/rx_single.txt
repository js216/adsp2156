# Slave single-lane RX control: FT4222 SingleWrite sends 32 bytes
# of 0xAA55AA55 over MOSI; DSP captures; cksum=0 (XOR of 8 identical
# words is zero).  Proves the master SingleWrite path drives CS+SCK
# and DSP slave single RX pipeline works end-to-end.

dsp:reset
dsp:uart_open
dsp:boot ldr=@main.ldr
delay ms=500
dsp:uart_write data="?\r"
delay ms=200
dsp:uart_write data="a 0\r"
delay ms=50
mark tag=rx_start
dsp:qspi_write data=@data.bin clk_div=8 mode=1 chunk_size=65535
mark tag=rx_end
delay ms=50
dsp:uart_write data="i\r"
dsp:uart_expect sentinel="sum=0x00000000" timeout_ms=3000
dsp:uart_close
