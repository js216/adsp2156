# 3-frame PRBS: n=128, chunk=64, prefix=1B → 3 CS frames (65,65,?).
dsp:reset
dsp:uart_open
dsp:boot ldr=@main.ldr
delay ms=500
dsp:uart_write data="m4\r"
delay ms=50
dsp:uart_write data="b 128\r"
delay ms=50
dsp:uart_write data="?\r"
delay ms=200
dsp:uart_write data="h 128\r"
delay ms=200
mark tag=s
dsp:qspi_write_prbs seed=0xC0FFEE n=128 clk_div=8 mode=4 chunk_size=64 prefix=@safety.bin
mark tag=e
dsp:uart_expect sentinel="w0000001f=" timeout_ms=5000
dsp:uart_close
