# Ping-pong RX demo.  Boots qspi firmware, switches to quad slave,
# primes the startup-race workaround, sends PRBS via master QSPI,
# asks DSP for checksum, exits when sum= appears.
#
# Blobs expected:
#   main.ldr   -- firmware with ping-pong ring
#   data.bin   -- payload bytes; length drives test size
#                 (256 KiB / 512 KiB / 1 MiB for 1/2/4-half tests)
#
# Example:
#   python3 submit.py plans/pp_halves.txt \
#     --blob main.ldr=qspi/main.ldr \
#     --blob data.bin=blobs/pp_1h_wire.bin \
#     --wait 30 --extract /tmp/out

dsp:reset
dsp:uart_open
dsp:boot ldr=@main.ldr
delay ms=500
# Lane width + auto-consume off + startup-race filler.
dsp:uart_write data="m4\r"
delay ms=50
dsp:uart_write data="a 0\r"
delay ms=50
dsp:uart_write data="?\r"
delay ms=200
mark tag=bursts_start
# One blob of PRBS-free payload; plugin writes it in a single pass.
# clk_div=8 (10 MHz SCK), mode=4 (quad).
dsp:qspi_write data=@data.bin clk_div=8 mode=4
mark tag=bursts_end
delay ms=100
dsp:uart_write data="i\r"
dsp:uart_expect sentinel="sum=0x" timeout_ms=5000
dsp:uart_close
