# Toggle PA_00 (MISO) via GPIO at ~5 Hz for ~2 s, then capture scope.
# Each g cmd takes ~10 ms round-trip so 10 toggle pairs = ~2 s.
dsp:reset
dsp:uart_open
dsp:boot ldr=@main.ldr
delay ms=500
dsp:uart_write data="g 0 0\r"
delay ms=100
dsp:uart_write data="g 0 1\r"
delay ms=100
dsp:uart_write data="g 0 0\r"
delay ms=100
dsp:uart_write data="g 0 1\r"
delay ms=100
dsp:uart_write data="g 0 0\r"
delay ms=100
dsp:uart_write data="g 0 1\r"
delay ms=100
dsp:uart_write data="g 0 0\r"
delay ms=100
dsp:uart_write data="g 0 1\r"
delay ms=100
dsp:uart_write data="g 0 0\r"
delay ms=100
dsp:uart_write data="g 0 1\r"
delay ms=100
scope:capture chans="C1,C2,C3,C4"
dsp:uart_close
