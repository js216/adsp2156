# Slave single-lane TX test. DSP pushes 16 bytes PRBS(0xC0FFEE)
# into TFIFO; master SingleRead clocks them back via MISO.
dsp:reset
dsp:uart_open
dsp:boot ldr=@main.ldr
delay ms=500
dsp:uart_write data="m1\r"
delay ms=50
dsp:uart_write data="?\r"
delay ms=200
dsp:uart_write data="t 0xC0FFEE 16\r"
dsp:uart_expect sentinel="tx queued" timeout_ms=2000
delay ms=100
mark tag=read
dsp:qspi_read n=16 clk_div=8 mode=1 chunk_size=16
mark tag=read_end
delay ms=50
dsp:uart_close
