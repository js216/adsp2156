# Same 128 PRBS bytes, sent via qspi_write (blob path), n=128, chunk=64,
# prefix=@safety.bin for fair compare vs qspi_write_prbs.
dsp:reset
dsp:uart_open
dsp:boot ldr=@main.ldr
delay ms=500
dsp:uart_write data="m4\r"
delay ms=50
dsp:uart_write data="b 128\r"
delay ms=50
dsp:uart_write data="?\r"
delay ms=200
dsp:uart_write data="h 128\r"
delay ms=200
mark tag=s
dsp:qspi_write data=@prbs128.bin clk_div=8 mode=4 chunk_size=64 prefix=@safety.bin
mark tag=e
dsp:uart_expect sentinel="w0000001f=" timeout_ms=5000
dsp:uart_close
